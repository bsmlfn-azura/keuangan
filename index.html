Koneksi Google Sheets Belum Terhubung

Hubungkan aplikasi ini dengan Google Spreadsheet Anda sebagai database. Setiap transaksi yang ditambahkan/diedit/dihapus akan otomatis tersinkronisasi ke spreadsheet.

Tempel URL dari deployment Google Apps Script Anda

Panduan Setup Google Sheets
1

Buat Google Spreadsheet

Buat spreadsheet baru di sheets.google.com. Di Sheet1, buat header baris pertama: id | date | type | category | description | amount

2

Buat Apps Script

Dari spreadsheet, klik Extensions → Apps Script. Hapus kode default, ganti dengan script yang disediakan di bawah. Ganti SPREADSHEET_ID dengan ID spreadsheet Anda dari URL.

3

Deploy sebagai Web App

Klik Deploy → New deployment. Pilih type Web app, Execute as Me, Who has access Anyone. Klik Deploy, lalu salin URL yang muncul.

4

Hubungkan ke DompetKu

Tempel URL web app di kolom di atas, lalu klik Simpan & Hubungkan. Data akan otomatis tersinkronisasi.

Kode Google Apps Script

const SPREADSHEET_ID = 'MASUKKAN_ID_SPREADSHEET';
const SHEET_NAME = 'Sheet1';

function getSheet() {
  return SpreadsheetApp.openById(SPREADSHEET_ID).getSheetByName(SHEET_NAME);
}

function doGet(e) {
  try {
    const sheet = getSheet();
    const data = sheet.getDataRange().getValues();
    if (data.length <= 1) return jsonResp({ status: 'ok', data: [] });
    const rows = data.slice(1).filter(r => r[0] !== '');
    const txs = rows.map(r => ({
      id: String(r[0]), date: fmtDate(r[1]),
      type: r[2], category: r[3],
      description: r[4], amount: Number(r[5])
    }));
    return jsonResp({ status: 'ok', data: txs });
  } catch (err) {
    return jsonResp({ status: 'error', message: err.message });
  }
}

function doPost(e) {
  try {
    const body = JSON.parse(e.postData.contents);
    if (body.action === 'add') return addRow(body.data);
    if (body.action === 'update') return updateRow(body.data);
    if (body.action === 'delete') return deleteRow(body.id);
    if (body.action === 'clear') return clearAll();
    return jsonResp({ status: 'error', message: 'Unknown action' });
  } catch (err) {
    return jsonResp({ status: 'error', message: err.message });
  }
}

function addRow(d) {
  const id = Utilities.getUuid();
  getSheet().appendRow([id, d.date, d.type, d.category, d.description, d.amount]);
  return jsonResp({ status: 'ok', id: id });
}

function updateRow(d) {
  const rows = getSheet().getDataRange().getValues();
  for (let i = 1; i < rows.length; i++) {
    if (String(rows[i][0]) === String(d.id)) {
      const s = getSheet();
      s.getRange(i+1,2).setValue(d.date);
      s.getRange(i+1,3).setValue(d.type);
      s.getRange(i+1,4).setValue(d.category);
      s.getRange(i+1,5).setValue(d.description);
      s.getRange(i+1,6).setValue(d.amount);
      return jsonResp({ status: 'ok' });
    }
  }
  return jsonResp({ status: 'error', message: 'Not found' });
}

function deleteRow(id) {
  const rows = getSheet().getDataRange().getValues();
  for (let i = 1; i < rows.length; i++) {
    if (String(rows[i][0]) === String(id)) {
      getSheet().deleteRow(i + 1);
      return jsonResp({ status: 'ok' });
    }
  }
  return jsonResp({ status: 'error', message: 'Not found' });
}

function clearAll() {
  const s = getSheet();
  if (s.getLastRow() > 1) s.deleteRows(2, s.getLastRow() - 1);
  return jsonResp({ status: 'ok' });
}

function fmtDate(v) {
  if (typeof v === 'string') return v;
  if (v instanceof Date) {
    return v.getFullYear()+'-'+
      String(v.getMonth()+1).padStart(2,'0')+'-'+
      String(v.getDate()).padStart(2,'0');
  }
  return String(v);
}

function jsonResp(d) {
  return ContentService.createTextOutput(JSON.stringify(d))
    .setMimeType(ContentService.MimeType.JSON);
}

Lokal (Browser)

Google Sheets

Data tersimpan di spreadsheet Google. Bisa diakses dari mana saja, bisa diedit langsung di Sheets.
